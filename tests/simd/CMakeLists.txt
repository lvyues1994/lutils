        foreach(isa IN ITEMS baseline advanced)
            add_library(simd_mixed_${mode}_${isa} OBJECT Mixed.cpp)
            target_link_libraries(simd_mixed_${mode}_${isa} PRIVATE lutils::simd lutils_development)
            if(mode STREQUAL "debug")
                target_compile_options(simd_mixed_${mode}_${isa} PRIVATE -O0)
            else()
                target_compile_options(simd_mixed_${mode}_${isa} PRIVATE -O3 -fno-inline)
            endif()
            if(isa STREQUAL "advanced")
                target_compile_options(simd_mixed_${mode}_${isa} PRIVATE -mavx2 -mfma)
                target_compile_definitions(simd_mixed_${mode}_${isa} PRIVATE LUTILS_MIXED_ADVANCED)
            else()
                target_compile_options(simd_mixed_${mode}_${isa} PRIVATE -msse2 -mno-avx -mno-fma)
            endif()
        endforeach()
        foreach(order IN ITEMS forward reverse)
            set(objects $<TARGET_OBJECTS:simd_mixed_${mode}_baseline> $<TARGET_OBJECTS:simd_mixed_${mode}_advanced>)
            if(order STREQUAL "reverse")
                list(REVERSE objects)
            endif()
            add_executable(simd_mixed_${mode}_${order} mixed_main.cpp ${objects})
            target_link_libraries(simd_mixed_${mode}_${order} PRIVATE lutils::simd_dispatch lutils_development)
            add_test(NAME simd.mixed.${mode}.${order} COMMAND simd_mixed_${mode}_${order})
            set_tests_properties(simd.mixed.${mode}.${order} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
        endforeach()
    endforeach()
endif()
