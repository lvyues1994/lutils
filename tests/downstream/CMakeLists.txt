cmake_minimum_required(VERSION 3.21)
project(lutils_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(SOURCE_ROOT)
    set(build_tools ${WITH_SHADER})
    if(LUTILS_HOST_KERNELC)
        set(build_tools OFF)
    endif()
    set(LUTILS_BUILD_KERNELC ${build_tools} CACHE BOOL "")
    set(LUTILS_BUILD_IMAGE_OPS ${WITH_OPS} CACHE BOOL "")
    set(LUTILS_ENABLE_VULKAN ${WITH_GPU} CACHE BOOL "")
    add_subdirectory("${SOURCE_ROOT}" lutils)
    if(LUTILS_BUILD_TESTS OR LUTILS_BUILD_EXAMPLES OR LUTILS_INSTALL)
        message(FATAL_ERROR "Embedded project enabled development/install targets by default")
    endif()
else()
    set(components ${REQUEST_COMPONENT})
    if(WITH_OPS)
        list(APPEND components image_ops)
    endif()
    if(WITH_SHADER)
        list(APPEND components kernelc)
    endif()
    if(WITH_GPU)
        list(APPEND components vulkan)
    endif()
    find_package(lutils 0.2 CONFIG REQUIRED COMPONENTS ${components} OPTIONAL_COMPONENTS ${OPTIONAL_COMPONENT})
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE lutils::core lutils::erasure lutils::image_memory lutils::compute_cpu)
if(WITH_OPS)
    target_link_libraries(consumer PRIVATE lutils::image_ops)
    target_compile_definitions(consumer PRIVATE WITH_OPS)
endif()
if(WITH_SHADER)
    lutils_add_shader(consumer_shader NAME consumer_shader SOURCE Kernel.hpp ENTRY consumer::Add
        INCLUDE_DIRS "private headers")
    target_link_libraries(consumer PRIVATE consumer_shader)
    target_compile_definitions(consumer PRIVATE WITH_SHADER)
endif()
if(WITH_GPU)
    target_link_libraries(consumer PRIVATE lutils::compute_vulkan)
    target_compile_definitions(consumer PRIVATE WITH_GPU)
endif()
