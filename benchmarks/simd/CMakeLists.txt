add_executable(simd_benchmark main.cpp Scalar.cpp)
target_link_libraries(simd_benchmark PRIVATE lutils::image_simd lutils_development)
# Keep the timed call opaque across TUs; never propagate ISA flags to the harness.
set_property(TARGET simd_benchmark PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
target_compile_definitions(simd_benchmark PRIVATE LUTILS_BENCH_BUILD="$<CONFIG>")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_sources(simd_benchmark PRIVATE Sse2.cpp Avx2.cpp)
    target_compile_definitions(simd_benchmark PRIVATE LUTILS_SIMD_X86)
    set_source_files_properties(Sse2.cpp PROPERTIES COMPILE_OPTIONS "-msse2;-mno-avx;-mno-fma")
    set_source_files_properties(Avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mno-fma")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    target_sources(simd_benchmark PRIVATE Neon.cpp)
    target_compile_definitions(simd_benchmark PRIVATE LUTILS_SIMD_ARM64)
endif()
