cmake_minimum_required(VERSION 3.21)
project(lutils VERSION 0.2.0 LANGUAGES CXX)
include(GNUInstallDirs)

set(lutils_top_level OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(lutils_top_level ON)
endif()

option(LUTILS_BUILD_TESTS "Build lutils tests" ${lutils_top_level})
option(LUTILS_BUILD_EXAMPLES "Build lutils examples" ${lutils_top_level})
option(LUTILS_BUILD_BENCHMARKS "Build image conversion benchmarks (requires kernelc)" OFF)
option(LUTILS_INSTALL "Install the CMake SDK and developer tools" ${lutils_top_level})
option(LUTILS_ENABLE_SANITIZERS "Instrument lutils tests and examples" OFF)
option(LUTILS_ENABLE_TSAN "Check CPU execution with ThreadSanitizer" OFF)
option(LUTILS_BUILD_KERNELC "Build the Clang 18 kernel compiler and image operations" OFF)
option(LUTILS_ENABLE_VULKAN "Build the Vulkan compute backend" OFF)
if(LUTILS_INSTALL AND (LUTILS_ENABLE_SANITIZERS OR LUTILS_ENABLE_TSAN))
    message(FATAL_ERROR "Sanitizer builds are for validation; use -DLUTILS_INSTALL=OFF or sdk-release")
endif()
if(LUTILS_ENABLE_SANITIZERS AND LUTILS_ENABLE_TSAN)
    message(FATAL_ERROR "ThreadSanitizer and AddressSanitizer require separate builds")
endif()

add_subdirectory(erasure)

add_library(lutils_development INTERFACE)
if(LUTILS_BUILD_TESTS OR LUTILS_BUILD_EXAMPLES OR LUTILS_BUILD_BENCHMARKS)
    target_compile_features(lutils_development INTERFACE cxx_std_17)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(lutils_development INTERFACE
            -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion
            -Wold-style-cast -Wnon-virtual-dtor)
        if(LUTILS_ENABLE_SANITIZERS)
            target_compile_options(lutils_development INTERFACE
                -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(lutils_development INTERFACE
                -fsanitize=address,undefined)
        elseif(LUTILS_ENABLE_TSAN)
            target_compile_options(lutils_development INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
            target_link_options(lutils_development INTERFACE -fsanitize=thread)
        endif()
    elseif(MSVC)
        target_compile_options(lutils_development INTERFACE /W4 /WX /permissive-)
    endif()
endif()

add_subdirectory(core)
add_subdirectory(image)
add_subdirectory(compute/kernel)
add_subdirectory(compute/runtime)
add_subdirectory(compute/backends/cpu)
add_subdirectory(image/memory)
if(LUTILS_ENABLE_VULKAN)
    add_subdirectory(compute/backends/vulkan)
endif()
if(LUTILS_BUILD_KERNELC)
    enable_language(C)
    add_subdirectory(tools/kernelc)
    set(LUTILS_KERNEL_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/compute/kernel/include")
    include(cmake/LutilsKernel.cmake)
    add_subdirectory(image/ops)
endif()

if(LUTILS_BUILD_KERNELC AND (LUTILS_BUILD_TESTS OR LUTILS_BUILD_EXAMPLES))
    add_subdirectory(examples/kernels)
endif()

if(LUTILS_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(LUTILS_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(LUTILS_BUILD_BENCHMARKS)
    if(NOT LUTILS_BUILD_KERNELC)
        message(FATAL_ERROR "LUTILS_BUILD_BENCHMARKS requires LUTILS_BUILD_KERNELC")
    endif()
    add_subdirectory(benchmarks)
endif()

if(LUTILS_INSTALL)
    include(cmake/LutilsInstall.cmake)
endif()
